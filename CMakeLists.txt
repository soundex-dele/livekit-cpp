cmake_minimum_required(VERSION 3.24)
project(livekit_cpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WEBRTC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../webrtc" CACHE PATH "Native WebRTC source checkout")
set(WEBRTC_PREBUILT_DIR "" CACHE PATH "Optional existing WebRTC CMake build (Windows Release only)")
find_package(OpenSSL REQUIRED)
find_package(Boost CONFIG REQUIRED COMPONENTS beast)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(cmake/WebRTC.cmake)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/protobuf EXCLUDE_FROM_ALL)
set(PROTO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/protocol/protobufs")
set(PROTO_NAMES logger/options livekit_metrics livekit_models livekit_rtc livekit_agent livekit_agent_dispatch livekit_egress livekit_room)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/logger")
foreach(name IN LISTS PROTO_NAMES)
  list(APPEND PROTO_SRCS "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.pb.cc")
  list(APPEND PROTO_HDRS "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.pb.h")
  list(APPEND PROTO_FILES "${PROTO_ROOT}/${name}.proto")
endforeach()
add_custom_command(OUTPUT ${PROTO_SRCS} ${PROTO_HDRS}
  COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR}/generated
    -I${PROTO_ROOT} -I${CMAKE_CURRENT_SOURCE_DIR}/third_party/protobuf/src ${PROTO_FILES}
  DEPENDS protobuf::protoc ${PROTO_FILES} VERBATIM)
add_library(livekit_protocol STATIC ${PROTO_SRCS})
target_include_directories(livekit_protocol PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(livekit_protocol PUBLIC protobuf::libprotobuf)
add_library(livekit_core STATIC src/auth.cpp src/config.cpp src/rtc.cpp src/rooms.cpp src/http.cpp)
target_include_directories(livekit_core PUBLIC include)
target_link_libraries(livekit_core PUBLIC livekit_protocol WebRTC::webrtc Boost::beast nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
target_compile_definitions(livekit_core PUBLIC BOOST_ASIO_NO_DEPRECATED)
if(MSVC)
  target_compile_options(livekit_core PRIVATE /W4 /utf-8 /bigobj)
  target_compile_definitions(livekit_core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(livekit-server src/main.cpp)
target_link_libraries(livekit-server PRIVATE livekit_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(livekit-tests tests/core_test.cpp)
  target_link_libraries(livekit-tests PRIVATE livekit_core)
  add_test(NAME core COMMAND livekit-tests)
  set_tests_properties(core PROPERTIES TIMEOUT 30)
endif()
install(TARGETS livekit-server RUNTIME DESTINATION bin)
