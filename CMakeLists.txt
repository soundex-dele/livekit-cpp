cmake_minimum_required(VERSION 3.24)
project(livekit_cpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  include(CheckCXXSourceCompiles)
  check_cxx_source_compiles("#include <optional>\nstruct Outer { struct Inner { int value = -1; }; std::optional<Inner> inner; };\nint main() { Outer value; value.inner.emplace(); }" LK_NATIVE_OPTIONAL_SUPPORTED)
  if(NOT LK_NATIVE_OPTIONAL_SUPPORTED)
    message(FATAL_ERROR "The compiler/standard-library combination cannot compile native WebRTC's nested optional types. Use the Linux preset with clang-15, libc++-15-dev and libc++abi-15-dev in a fresh build directory.")
  endif()
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WEBRTC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../webrtc" CACHE PATH "Native WebRTC source checkout")
set(WEBRTC_PREBUILT_DIR "" CACHE PATH "Optional existing WebRTC CMake build (Windows Release only)")
find_package(OpenSSL REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(cmake/WebRTC.cmake)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/protobuf EXCLUDE_FROM_ALL)
set(PROTO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/protocol/protobufs")
set(PROTO_NAMES logger/options livekit_metrics livekit_models livekit_rtc livekit_agent livekit_agent_dispatch livekit_egress livekit_room)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/logger")
foreach(name IN LISTS PROTO_NAMES)
  list(APPEND PROTO_SRCS "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.pb.cc")
  list(APPEND PROTO_HDRS "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.pb.h")
  list(APPEND PROTO_FILES "${PROTO_ROOT}/${name}.proto")
endforeach()
add_custom_command(OUTPUT ${PROTO_SRCS} ${PROTO_HDRS}
  COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR}/generated
    -I${PROTO_ROOT} -I${CMAKE_CURRENT_SOURCE_DIR}/third_party/protobuf/src ${PROTO_FILES}
  DEPENDS protobuf::protoc ${PROTO_FILES} VERBATIM)
add_library(livekit_protocol STATIC ${PROTO_SRCS})
target_include_directories(livekit_protocol PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(livekit_protocol PUBLIC protobuf::libprotobuf)
add_library(livekit_core STATIC src/auth.cpp src/config.cpp src/turn.cpp src/rtc.cpp src/codec_audit.cpp src/native_rtp.cpp src/native_send_transport.cpp src/sfu.cpp src/rooms.cpp src/http.cpp)
if(WIN32)
  # The pinned manifest defaults to stdlib queues on every platform. Compile
  # and select libwebrtc's Windows timer implementation without changing it.
  target_sources(livekit_core PRIVATE "${WEBRTC_ROOT}/rtc_base/task_queue_win.cc")
endif()
target_include_directories(livekit_core PUBLIC include)
target_link_libraries(livekit_core PUBLIC livekit_protocol WebRTC::webrtc Boost::headers nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
target_compile_definitions(livekit_core PUBLIC BOOST_ASIO_NO_DEPRECATED)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # Boost 1.74 detects invoke_result only for GCC; libc++ removes result_of
  # in C++20. All consumers of our Asio-bearing headers use C++20 as well.
  target_compile_definitions(livekit_core PUBLIC BOOST_ASIO_HAS_STD_INVOKE_RESULT=1)
endif()
if(MSVC)
  target_compile_options(livekit_core PRIVATE /W4 /utf-8 /bigobj)
  target_compile_definitions(livekit_core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(livekit-server src/main.cpp)
target_link_libraries(livekit-server PRIVATE livekit_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(livekit-turn-tests tests/turn_credentials_test.cpp)
  target_link_libraries(livekit-turn-tests PRIVATE livekit_core)
  add_test(NAME turn-credentials COMMAND livekit-turn-tests)
  set_tests_properties(turn-credentials PROPERTIES TIMEOUT 30)
  add_executable(livekit-native-send-tests tests/native_send_transport_test.cpp)
  target_link_libraries(livekit-native-send-tests PRIVATE livekit_core)
  add_test(NAME native-send COMMAND livekit-native-send-tests)
  set_tests_properties(native-send PROPERTIES TIMEOUT 30)
  add_executable(livekit-native-rtp-tests tests/native_rtp_test.cpp)
  target_link_libraries(livekit-native-rtp-tests PRIVATE livekit_core)
  add_test(NAME native-rtp COMMAND livekit-native-rtp-tests)
  set_tests_properties(native-rtp PROPERTIES TIMEOUT 30)
  add_executable(livekit-tests tests/core_test.cpp)
  target_link_libraries(livekit-tests PRIVATE livekit_core)
  add_test(NAME core COMMAND livekit-tests)
  set_tests_properties(core PROPERTIES TIMEOUT 30)
endif()
install(TARGETS livekit-server RUNTIME DESTINATION bin)
